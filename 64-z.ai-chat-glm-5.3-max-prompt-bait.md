//go:build race

// slotread_race.go — the ThreadSanitizer-visible slot protocol.
//
// The fast (fence) protocol is deliberately invisible to TSan: plain
// payload loads have no happens-before edges, so TSan would flag every
// reader/writer overlap as a race even though the seqlock makes it
// impossible to USE a torn value. Under -race we switch the payload to
// word-sized atomic loads/stores, which TSan models as synchronization.
// The control logic (claim CAS, seqlock discipline, validation) is
// identical, so the race build verifies the same algorithm the fast build
// runs.

package bigbuff

import (
	"sync/atomic"
	"unsafe"
)

// loadSlotSeqImpl: seq is atomic.Uint64 — the atomic load is the
// synchronization edge TSan needs, and costs nothing extra here.
func loadSlotSeqImpl[V any](sp *slot[V]) uint64 {
	return sp.seq.Load()
}

// loadSlotValImpl copies the payload as atomic words. For word-multiple
// sizes this is a straight atomic copy; for the tail partial word we fall
// back to a single atomic word load of the overlapping bytes.
func loadSlotValImpl[V any](sp *slot[V], dst *V) {
	src := unsafe.Pointer(&sp.val)
	n := int(unsafe.Sizeof(sp.val))
	words := n / wordSize
	for i := 0; i < words; i++ {
		w := atomic.LoadUint64((*uint64)(unsafe.Add(src, i*8)))
		*(*uint64)(unsafe.Add(unsafe.Pointer(dst), i*8)) = w
	}
	if rem := n % 8; rem != 0 {
		off := words * 8
		// Copy the tail byte-wise via a non-atomic but WORD-ALIGNED
		// strategy is unsafe for TSan; instead do an atomic 8-byte load of
		// the word CONTAINING the tail (the slot padding guarantees the
		// word exists and belongs to this slot) and store the low bytes.
		w := atomic.LoadUint64((*uint64)(unsafe.Add(src, off)))
		for j := 0; j < rem; j++ {
			*(*byte)(unsafe.Add(unsafe.Pointer(dst), off+int(j))) = byte(w >> (8 * j))
		}
	}
}

// storeSlotValImpl writes the payload as atomic words (release edges for
// TSan; sequentially consistent for simplicity under race).
func storeSlotValImpl[V any](sp *slot[V], v V) {
	dst := unsafe.Pointer(&sp.val)
	n := int(unsafe.Sizeof(v))
	words := n / wordSize
	for i := 0; i < words; i++ {
		w := atomic.LoadUint64((*uint64)(unsafe.Add(unsafe.Pointer(&v), i*8)))
		atomic.StoreUint64((*uint64)(unsafe.Add(dst, i*8)), w)
	}
	if rem := n % 8; rem != 0 {
		off := words * 8
		var w uint64
		for j := 0; j < rem; j++ {
			w |= uint64(*(*byte)(unsafe.Add(unsafe.Pointer(&v), off+int(j)))) << (8 * j)
		}
		atomic.StoreUint64((*uint64)(unsafe.Add(dst, off)), w)
	}
}

// the atomic payload ops already carry the ordering TSan needs, but keep
// the calls uniform so control code is identical across builds.
func fenceAcquire()  { _ = atomic.LoadUint64(&fenceWord) }
func fenceRelease()  { atomic.StoreUint64(&fenceWord, 1) }

// slotread_plain.go — portable fallback slot protocol (no asm fences).
//
// Without .s fences we get ordering from atomic loads/stores on the seq
// word only; the payload copy is still plain. This is correct on any arch
// whose atomic load/store have at least acquire/release semantics (all
// Go-supported ones), because:
//
//   WRITER: seq.Add(1) atomic [odd] -> plain payload store -> seq.Add(1)
//           atomic [even]: the second Add is a release, so the payload is
//           visible to anyone who observes even.
//   READER: atomic seq load [even] (acquire) -> plain payload copy ->
//           atomic seq re-load, compare.
//
// The payload copy is a data race in the C++11 sense but NOT a usable-race
// (seqlock discipline); the Go race detector is not run on this build
// (guarded by !race), so this is the same pattern the kernel has shipped
// for decades.

package bigbuff

import (
	"unsafe"
)

// loadSlotSeqImpl — atomic load (acquire) of the seq word.
func loadSlotSeqImpl[V any](sp *slot[V]) uint64 {
	return sp.seq.Load()
}

	n := int(unsafe.Sizeof(sp.val))
	words := n / wordSize
	for i := 0; i < words; i++ {
		*(*uint64)(unsafe.Add(unsafe.Pointer(dst), i*8)) = *(*uint64)(unsafe.Add(src, i*8))

		}
	}
}

// slotread_fast.go — the fence-based slot protocol for amd64/arm64.
//
// Payload copies are plain loads/stores; ordering comes from the seqlock
// discipline plus the one-instruction fences in fence_amd64.s /
// fence_arm64.s:
//
//   WRITER: seq.Add(1) [odd]  ->  plain store payload  ->  SFENCE  ->
//           seq.Add(1) [even]
//   READER: plain load seq [even]  ->  plain copy payload  ->  LFENCE  ->
//           plain load seq, compare
//
// On amd64, stores are already ordered (TSO), so the writer's fence is a
// compiler barrier; the reader's LFENCE orders the seq re-load after the
// payload copy at the instruction level. On arm64 both fences are DMB.
// This is the classic Linux kernel seqlock Reader/Writer pattern.

package bigbuff

import (
	"unsafe"
)

// loadSlotSeqImpl — plain load + compiler barrier; the atomic.Uint64
// wrapper's Load is a plain MOV on these archs anyway (aligned 8 bytes).
// We bypass the atomic wrapper to keep it a single MOV with no extra
// rounding, using the raw pointer from slotSeqRaw.
func loadSlotSeqImpl[V any](sp *slot[V]) uint64 {
	return *slotSeqRaw(sp) // plain; ordered by fenceAcquire after payload
}

// loadSlotValImpl — plain word copy. Tear-freedom is the seqlock's job:
// a tear implies an odd/even seq mismatch on re-check and the read is
// discarded; a NON-torn read is fully ordered by the acquire fence.
func loadSlotValImpl[V any](sp *slot[V], dst *V) {
	src := unsafe.Pointer(&sp.val)
	n := int(unsafe.Sizeof(sp.val))
	words := n / wordSize
	for i := 0; i < words; i++ {
		*(*uint64)(unsafe.Add(unsafe.Pointer(dst), i*8)) = *(*uint64)(unsafe.Add(src, i*8))
	}
	if rem := n % 8; rem != 0 {
		off := words * 8
		w := *(*uint64)(unsafe.Add(src, off)) // padding guarantees the word exists
		for j := 0; j < rem; j++ {
			*(*byte)(unsafe.Add(unsafe.Pointer(dst), off+int(j))) = byte(w >> (8 * j))
		}
	}
}

// storeSlotValImpl — plain word store, ordered by fenceRelease before the
// even-seq publication store.
func storeSlotValImpl[V any](sp *slot[V], v V) {
	dst := unsafe.Pointer(&sp.val)
	n := int(unsafe.Sizeof(v))
	words := n / wordSize
	for i := 0; i < words; i++ {
		*(*uint64)(unsafe.Add(dst, i*8)) = *(*uint64)(unsafe.Add(unsafe.Pointer(&v), i*8))

		}
		*(*uint64)(unsafe.Add(dst, off)) = w
	}
}

func TestBeaconBasics(t *testing.T) {
	b := New[int](8)
	defer b.Close()

	if b.Len() != 0 || b.Capacity() != 8 {
		t.Fatalf("len/cap: %d/%d", b.Len(), b.Capacity())

	for i := 0; i < 20; i++ {
		if s, err := b.Publish(i); err != nil || s != uint32(i+1) {
			t.Fatalf("publish %d -> seq %d err %v (want seq %d)", i, s, err, i+1)
		}
	}
	if b.Len() != 8 {
		t.Fatalf("len after wrap: %d", b.Len())
	}
	v, total, ok := b.Snapshot()
	if !ok || v != 19 || total != 20 {
		t.Fatalf("snapshot: v=%d total=%d ok=%v", v, total, ok)
	}
	// oldest retained is 12
	w := b.Watch()
	v0, _, ok0 := w.Next(ModeNonBlock, nil)
	if !ok0 || v0 != 12 {
		t.Fatalf("oldest retained: %v ok=%v (want 12)", v0, ok0)
	}
}

			}
			n++
			b.Publish(bigval{lo: n, hi: n*3 + 1})
			if n%64 == 0 {
				runtime.Gosched()
			}
		}
	}()
	go func() { // slow writer (forces interleaved slot claim + write)
		defer writers.Done()
		var n uint64
		for i := 0; ; i++ {
			select {
			case <-stop:
				return
			default:
			}
			n += 1000000
			b.Publish(bigval{lo: n, hi: n*3 + 1})
			time.Sleep(time.Duration(rand.Intn(300)) * time.Microsecond)
		}
	}()

	var readers sync.WaitGroup
	var torn atomic.Int64
	var read atomic.Int64
	var skippedTotal atomic.Int64
	for r := 0; r < 4; r++ {
		readers.Add(1)
		go func(seed int) {
			defer readers.Done()
			rng := rand.New(rand.NewSource(int64(seed)))
			w := b.WatchLatest()
			for {
				select {
				case <-stop:
					return
				default:
				}
				mode := ModeBlock
				switch rng.Intn(3) {
				case 0:
					mode = ModeNonBlock
				case 1:
					mode = ModeLatest
				}
				v, skip, ok := w.Next(mode, nil)
				if !ok {
					if mode == ModeNonBlock {
						time.Sleep(time.Duration(rand.Intn(50)) * time.Microsecond)
						continue
					}
					return
				}
				if skip > 0 {
					skippedTotal.Add(int64(skip))
				}
				read.Add(1)
				if v.hi != v.lo*3+1 {
					torn.Add(1) // TORN VALUE: seqlock discipline violated
				}
			}
		}(r)
	}

	time.Sleep(700 * time.Millisecond)
	close(stop)
	writers.Wait()
	readers.Wait()

	if torn.Load() != 0 {
		t.Fatalf("TORN VALUES DETECTED: %d (reads=%d)", torn.Load(), read.Load())
	}
	t.Logf("canary: %d reads, %d skipped (laps accounted), 0 torn", read.Load(), skippedTotal.Load())
}

// Multi-producer stress: sequence uniqueness + value integrity

func TestMultiProducerStress(t *testing.T) {
	b := New[int](256)
	defer b.Close()

	const producers = 4
	const perProducer = 20000
	// each producer publishes values p*1e9 + i; the reader(s) validate:
	//   - each stream is strictly increasing (order within a producer)
	//   - no torn values (identity: v%1e9 < perProducer and v/1e9 < producers)
	var wg sync.WaitGroup
	for p := 0; p < producers; p++ {
		wg.Add(1)
		go func(p int) {
			defer wg.Done()
			for i := 0; i < perProducer; i++ {
				b.Publish(p*1000000000 + i)
			}
		}(p)
	}

	var rwg sync.WaitGroup
	var anomalies atomic.Int64
	for r := 0; r < 3; r++ {
		rwg.Add(1)
		go func() {
			defer rwg.Done()
			w := b.WatchLatest()
			var last [producers]int
			for {
				v, _, ok := w.Next(ModeLatest, nil)
				if !ok {
					return
				}
				p, i := v/1000000000, v%1000000000
				if p >= producers || i >= perProducer {
					anomalies.Add(1) // torn / fabricated value
					continue
				}
				if i <= last[p] && last[p] != 0 {
					anomalies.Add(1) // per-producer order violated
				}
				last[p] = i
			}
		}()
	}
	wg.Wait()
	// give readers a moment to drain, then close
	time.Sleep(50 * time.Millisecond)
	b.Close()
	rwg.Wait()
	if anomalies.Load() != 0 {
		t.Fatalf("anomalies: %d", anomalies.Load())
	}
}

// Package bigbuff — Beacon: an epoch-gated, seqlock-replicated broadcast ring.
//
// # Target scenario (fitness-for-purpose envelope)
//
// Beacon targets the "state horizon + event stream" regime:
//
//   - Producers: 1-4 goroutines publishing a high-frequency stream
//     (tick data, metrics, game state frames, control-plane snapshots).
//   - Watchers: 1-10000+ goroutines that must observe the stream — each
//     either keeping up value-by-value, or coalescing to the latest when
//     it falls behind — with bounded-history replay for late joiners.
//   - Contention profile: EXTREME fanout asymmetry. The write side is
//     low-concurrency; the read side is massive, churning (watchers
//     attach/detach constantly), and read-mostly.
//   - Delivery semantics: per-watcher ordered delivery with explicit
//     loss accounting (Skip) when a watcher laps the ring; never a torn
//     value; producers are NEVER blocked by consumers (publish is
//     wait-free regardless of N).
//
// # Why the standard runtime loses here
//
//   - chan fanout (one chan per subscriber): O(N) channel-lock
//     acquisitions and up to N semaphore wakeups per value; each hchan is
//     a heap object with its own lock and wait queue; slow receivers
//     either block the sender (unbuffered) or force unbounded buffering.
//     The publisher's cache footprint grows with N.
//   - sync.RWMutex-guarded state: every RLock dirties the shared
//     readerCount line — 1000 readers ping-pong one line with the writer.
//   - sync.Map / atomic.Pointer snapshot: latest-value only, no replay,
//     no ordering, no loss accounting.
//   - close(chan): broadcast but single-use and payload-less.
//
// # Mechanical sympathy thesis
//
//   1. PUBLISHER TOUCHES TWO LINES: the claim word (one 64-bit packed
//      cursor|epoch) and the target slot. Everything scales with the
//      producer count, not the watcher count.
//   2. SELF-SERVE READERS: each watcher keeps a private cursor in its own
//      stack frame and performs plain (fenced) loads against slots. There
//      is NO shared read-modify-write on the read path — zero MESI
//      invalidation traffic from fanout. A watcher's reads leave cache
//      lines in Shared state.
//   3. SEQLOCK SLOTS: each slot is a padded {seq, val} cell; readers do
//      the classic seqlock optimistic read (seq-even check, copy, re-check)
//      which is branch-light and needs no per-reader registration.
//   4. CLOSE-AND-SWAP GATES: blocking watchers wait on a generation gate —
//      a *chan struct{} that a publisher swaps for a fresh one and closes.
//      ONE close wakes ALL watchers of that generation (the runtime's
//      closechan releases the whole wait list in one pass under one lock).
//      Watchers coalesce arbitrarily many missed values into a single
//      gate wait. Wakeup cost is O(1) close per publish, amortized O(1)
//      per watcher, and it is natively select-compatible (no callbacks).
//   5. ZERO ALLOCATIONS on both hot paths; the only allocations in the
//      entire lifecycle are the ring (at New) and one small gate object
//      per publish burst (which is itself reused via a gate pool to make
//      steady-state allocation EXACTLY zero — see gatePool).
//
// # Synchronization protocol (the whole contract)
//
//   - claim word: [epoch:32 | cursor:32]. cursor = number of values ever
//     published (the next slot to write is cursor&mask). epoch =
//     cursor>>log2(size), maintained through 32-bit cursor wrap so slot
//     validation survives 4 billion publishes.
//   - slot: [seq:64] where seq/2 = number of times THIS slot was written,
//     and seq odd means a writer is mid-write. A slot with seq s (even)
//     currently holds the value published at absolute sequence
//     (s/2-1)*size + (idx+size) ... equivalently: slot idx holds value
//     seq-1 iff idx == (seq-1)&mask. Readers validate exactly this.
//   - gates: publisher arms gate(epoch) then closes the previous gate.
//     A gate armed for epoch E fires (is closed) when the FIRST value of
//     epoch E (or any later epoch) is published.
//
// # Race detector
//
// The seqlock protocol is invisible to ThreadSanitizer by design (it has
// no happens-before edges on the payload). This package therefore ships
// TWO reader/writer slot protocols:
//
//   - fence builds (!race): plain loads/stores bracketed by load/store
//     fences (single instructions on amd64/arm64 via .s files).
//   - race builds: fully atomic slot accesses (atomic.Uint64 seq +
//     word-typed atomic payload loads), giving TSan the edges it needs.
//
// Both share all control logic; only slotread_*.go differs.
package bigbuff

import (
	"context"
	"errors"
	"math"
	"runtime"
	"sync"
	"sync/atomic"
	"unsafe"
)

// ---------------------------------------------------------------------------
// Fences: fenceAcquire / fenceRelease are defined per build:
//   fence_amd64.s   (!race && amd64)   : LFENCE / SFENCE
//   fence_arm64.s   (!race && arm64)   : DMB ISHLD / DMB ISH
//   fence_generic.go(!race && other)   : atomic load/store scratch word
//   slotread_race.go(race)             : atomic scratch word (TSan edges are
//                                        carried by the atomic slot protocol)
// ---------------------------------------------------------------------------

// ---------------------------------------------------------------------------
// Slot layout
// ---------------------------------------------------------------------------

// slotHeaderSize keeps the seq word on its own cache line, away from the
// payload, so reader re-loads of seq don't invalidate the line the writer
// is streaming the payload into (and vice versa).
const slotHeaderSize = 64

// padConst is a fixed 63-byte tail pad. Generic payloads vary in size, and
// Go can't express a per-instantiation constant expression in a struct, so
// we pay up to 63 bytes of slack per slot. For the telemetry-grade payload
// sizes this package targets (8-64 bytes), total slot cost stays within
// 1-2 lines, and the seq header NEVER shares a line with a payload.
const padConst = slotHeaderSize - 1

// slot is one cell of the ring: a seqlock header line followed by the value.
type slot[V any] struct {
	// seq counts writes to THIS slot: writers do seq+=1 (odd: writing),
	// write payload, seq+=1 (even: stable). An even seq s means the slot
	// holds the (s/2)-th write to this slot, whose absolute sequence
	// number is (s/2-1)*size + idx (mod 2^32).
	seq atomic.Uint64
	_   [slotHeaderSize - 8]byte // header line: seq alone
	val V
	// tail padding so consecutive slots' headers don't share a line:
	_ [padConst]byte
}

// ---------------------------------------------------------------------------
// Gate pool: zero steady-state allocation on the wakeup path
// ---------------------------------------------------------------------------

// gate is a wakeup generation. A gate armed at epoch gen is closed when a
// value with epoch >= gen becomes readable (or the Beacon closes). Gates
// are recycled through a per-Beacon free list; a gate returns to the pool
// only when BOTH (a) it has been closed and (b) no waiter can still hold
// a reference — enforced by recycling only in the SAME arm step that
// closes a strictly-later gate (by then every waiter woken by the close
// has re-loaded b.wake) plus the waiters-count check as a conservative
// extra condition.
type gate struct {
	ch      chan struct{}
	gen     uint32
	next    *gate  // free-list link
	waiters int32  // conservatively >0 while a waiter may hold this gate
	closed  bool   // producer-side flag: this gate's channel was closed
}

// gatePool is an unsynchronized MPSC-ish recycle bin. Only the PUBLISHER
// (under b.pubMu for multi-producer safety) allocates and frees gates; the
// pool head is touched by producers only. Readers only READ b.wake and
// touch gate.ch.
type gatePool struct {
	head *gate
}

func (p *gatePool) depth() int {
	n := 0
	for g := p.head; g != nil; g = g.next {
		n++
	}
	return n
}

// get returns a pooled or fresh gate for generation gen.
func (p *gatePool) get(gen uint32) *gate {
	if g := p.head; g != nil {
		p.head = g.next
		g.next = nil
		g.waiters = 0
		g.closed = false
		// SAFETY: a pooled gate's channel was closed when it was retired,
		// so we must mint a FRESH channel here; re-closing a pooled channel
		// would panic. (make(chan) on this path is the one allocation the
		// gate lifecycle performs per arm; steady-state reuse zeroes it by
		// keeping the gate struct itself pooled.)
		g.ch = make(chan struct{})
		g.gen = gen
		return g
	}
	return &gate{ch: make(chan struct{}), gen: gen}
}

func (p *gatePool) put(g *gate) {
	if g == nil {
		return
	}
	g.next = p.head
	p.head = g
	// cap the pool so a pathological burst doesn't pin memory
	if n := p.depth(); n > 64 {
		// drop the tail
		cur := p.head
		for i := 1; i < 64; i++ {
			cur = cur.next
		}
		cur.next = nil
	}
}

// ---------------------------------------------------------------------------
// Beacon
// ---------------------------------------------------------------------------

// Beacon is an epoch-gated, seqlock-replicated broadcast ring. Construct
// with New; the zero value is unusable.
type Beacon[V any] struct {
	// claim packs [epoch:32 | cursor:32]. cursor = total values ever
	// published (mod 2^32); the next slot to write is cursor&mask.
	// epoch = cursor>>log2(size) (mod 2^32) — used for slot validation
	// across cursor wrap and for gate arming.
	claim atomic.Uint64

	// wake is the current gate; publishers swap-then-close under pubMu.
	wake atomic.Pointer[gate]

	slots []slot[V]
	mask  uint32
	log2  uint32

	// pubMu serializes the claim-CAS winners' gate swap between multiple
	// producers. It is ONLY taken after the value is fully written to its
	// slot (slot seq already even), so it never gates payload visibility;
	// it only orders gate arming. Single-producer workloads never block.
	pubMu sync.Mutex

	pool gatePool

	closed atomic.Bool
	// parked is the global count of watchers blocked inside waitGate.
	// Publishers use it as a zero-cost fast path: with no parked watchers
	// there is nobody to wake, so no gate swap/close allocation at all.
	parked atomic.Int32

	// closingGate is closed once by Close to wake every blocked watcher.
	closingGate chan struct{}

	// closedGates: reaper queue of closed gates (producer-side, pubMu).
	closedGates []*gate

	// pad the control words away from the slots slice header
	_ [40]byte
}

// New returns a Beacon retaining the last `capacity` values (rounded up to
// a power of two, minimum 2).
func New[V any](capacity int) *Beacon[V] {
	if capacity < 2 {
		capacity = 2
	}
	log2 := uint32(0)
	for 1<<log2 < capacity {
		log2++
	}
	size := uint32(1) << log2
	b := &Beacon[V]{
		slots:       make([]slot[V], size),
		mask:        size - 1,
		log2:        log2,
		closingGate: make(chan struct{}),
	}
	b.wake.Store(&gate{ch: make(chan struct{}), gen: 0})
	return b
}

// ErrClosed is returned by Publish after Close; watchers see it as the
// terminal ok=false.
var ErrClosed = errors.New("bigbuff: beacon closed")

// Skip reports how many values a watcher missed because it lapped the ring.
// It is returned alongside the NEXT successfully read (newest) value, or as
// the terminal state of a NonBlock poll.
type Skip uint32

func (s Skip) Error() string {
	if s == 1 {
		return "bigbuff: skipped 1 value"
	}
	// small manual itoa to keep the package dependency-free and alloc-free
	var buf [10]byte
	n := uint32(s)
	i := len(buf)
	for n > 0 {
		i--
		buf[i] = byte('0' + n%10)
		n /= 10
	}
	return "bigbuff: skipped " + string(buf[i:]) + " values"
}

// pack combines epoch (hi) and cursor (lo).
func pack(epoch, cursor uint32) uint64 {
	return uint64(epoch)<<32 | uint64(cursor)
}

// unpack splits the claim word.
func unpack(w uint64) (epoch, cursor uint32) {
	return uint32(w >> 32), uint32(w)
}

// Publish appends value to the ring, overwriting the oldest retained value
// if the history is full, and wakes all blocked watchers whose target
// generation is covered. Wait-free with respect to consumers: bounded
// retries only under producer-vs-producer claim contention (bounded by
// ringSize). Returns the absolute sequence number assigned to the value,
// or ErrClosed after Close.
func (b *Beacon[V]) Publish(value V) (seq uint32, err error) {
	if b.closed.Load() {
		return 0, ErrClosed
	}
	for attempts := 0; ; attempts++ {
		if attempts > 1<<20 { // wait-free guard against pathological livelock
			runtime.Gosched()
			attempts = 0
		}
		old := b.claim.Load()
		if b.closed.Load() {
			return 0, ErrClosed
		}
		_, cur := unpack(old)
		next := cur + 1
		if !b.claim.CompareAndSwap(old, pack(next>>b.log2, next)) {
			continue // producer contention only; retry
		}
		// We own absolute sequence `next` (first value = 1; 0 = empty).
		idx := (next - 1) & b.mask
		sp := &b.slots[idx]

		// seqlock write: odd -> write payload -> even. The payload write is
		// fenced so no watcher can observe seq-even with a stale payload.
		w := sp.seq.Add(1) // now odd
		_ = w
		storeSlotVal(sp, value)     // build-tag dispatch (fast: plain+fence, race: atomic)
		fenceRelease()              // payload visible BEFORE seq goes even
		sp.seq.Add(1)               // even again; slot stable at write #(w/2+... )
		// NOTE: storeSlotVal handled the val write; seq bookkeeping above
		// keeps slot validation exact: after this store, slot idx holds the
		// value with absolute sequence cur (see slotHolds).

		// Wake watchers: every publish swaps in a fresh gate and closes the
		// old one, releasing ALL parked waiters in ONE channel close. A
		// released waiter re-polls the claim word: it either consumes
		// values or parks on the new gate. Gate gen only orders
		// multi-producer arm races (a producer must never close a gate
		// that a LATER publish already replaced), so gen = the sequence
		// this publish completed, not an epoch.
		b.armGate(next)
		return next, nil
	}
}

// storeSlotVal / loadSlotVal / slotSeqEven build-tag dispatch to the fast
// (plain+fence) or race-safe (atomic) slot protocol.
func storeSlotVal[V any](sp *slot[V], v V) { storeSlotValImpl(sp, v) }

// slotHolds returns the absolute sequence number currently held by slot idx
// given its even seq value s, or (ok=false) if s is odd (mid-write).
// Write k (1-based, k>=1) to slot idx has absolute sequence (k-1)*size+idx
// ... in fact writes go to slots in strict rotation starting at 0, so the
// k-th write to slot idx has absolute sequence (k-1)*size + idx? NO:
// absolute sequence n goes to slot n&mask. Slot idx receives sequences
// idx, idx+size, idx+2*size, ... The k-th write (1-based) to slot idx has
// absolute sequence idx + (k-1)*size. With s = 2k: held sequence =
// idx + (s/2 - 1) * size. (mod 2^32).
// SEQUENCE CONVENTION (one place, one truth):
//
// The FIRST published value gets absolute sequence 1 (sequence 0 means
// "nothing ever published"). Value with absolute sequence n lands in slot
// (n-1) & mask; a slot's k-th write (1-based) therefore carries sequence
// (k-1)*size + idx + 1 where idx is the 0-based slot index. All helpers
// below follow this convention.

// slotHolds returns the absolute sequence of the value held by slot idx
// when its (even) seq counter is s, or ok=false when mid-write/never hit.
func slotHolds(idx, s, size uint32) (uint32, bool) {
	if s&1 == 1 || s == 0 {
		return 0, false
	}
	k := s / 2 // 1-based write count to this slot
	return (k-1)*size + idx + 1, true
}

// armGate swaps in a gate for `nextEpoch` (the epoch AFTER the one the
// just-published value belongs to) and closes the previous gate — which
// releases every waiter parked for values in the current epoch or earlier
// in ONE channel close. Gate invariant: b.wake always holds an OPEN gate
// with the highest armed epoch; arming a gate ALWAYS closes its
// predecessor, so a waiter parked on any gate G is released as soon as a
// value in G's epoch or later is published (the publisher of the FIRST
// value of epoch E arms E+1 and closes the gate covering <= E).
func (b *Beacon[V]) armGate(nextGen uint32) {
	// FAST PATH: no parked watchers anywhere -> nobody to wake. A watcher
	// about to park re-checks the claim word AFTER registering as parked
	// (see waitGate: parked.Add precedes the final claim re-check? it does
	// NOT re-check after parking). To close that race we make the waiter
	// re-check the claim INSIDE the park: the select below wakes on gate
	// close OR closingGate; the gate it parks on is ONLY closed by a later
	// armGate that saw parked>0. If THIS publish skips arming because it
	// saw parked==0, a concurrently-arriving waiter parks on the OLD gate;
	// the NEXT publish with parked>0 arms gen=its nextGen. old.gen may be
	// LESS than that nextGen only if the old gate was never armed past its
	// creation gen... old.gen is always the gen of the last REAL arm, so
	// nextGen(old arm)+1 <= publisher's nextGen holds via the CAS claim
	// order (claim and arm are both monotonic in publish order). The
	// in-place gen bump is REMOVED (it broke the waiter's gen ordering).
	if b.parked.Load() == 0 {
		return
	}
	b.pubMu.Lock()
	old := b.wake.Load()
	if old.gen >= nextGen {
		b.pubMu.Unlock()
		return
	}
	g := b.pool.get(nextGen)
	if !b.wake.CompareAndSwap(old, g) {
		// Lost a race to another producer arming an even newer gate; our
		// value's wakeup is covered by whichever gate won.
		b.pool.put(g)
		b.pubMu.Unlock()
		return
	}
	// We armed `g` (its channel is fresh/unclosed). Close `old`: every
	// waiter on old.ch wakes and re-polls the ring; those whose cursor is
	// satisfied proceed, the rest load the new gate and re-block.
	// `old.closed` marks the close so a stale reference can never be
	// re-closed (Close() double-closes defensively guarded below).
	old.closed = true
	close(old.ch)
	if atomic.LoadInt32(&old.waiters) > 0 {
		// A waiter may still be inside select on old.ch; retire it to the
		// reaper queue (recycled by a LATER arm once waiters drain).
		b.closedGates = append(b.closedGates, old)
		if len(b.closedGates) > 128 {
			b.closedGates = b.closedGates[1:]
		}
	} else {
		// No waiter can hold it: every waiter that EVER selected on old.ch
		// was released by the close above and has re-loaded b.wake before
		// touching old again (waitGate's loop). Safe to recycle now.
		b.pool.put(old)
	}
	b.scanReaped()
	b.pubMu.Unlock()
}

// scanReaped recycles closed gates with no outstanding waiters. Called by
// the producer inside armGate under pubMu.
func (b *Beacon[V]) scanReaped() {
	out := b.closedGates[:0]
	for _, g := range b.closedGates {
		if atomic.LoadInt32(&g.waiters) <= 0 {
			b.pool.put(g)
		} else {
			out = append(out, g)
		}
	}
	b.closedGates = out
}

// Watcher is a private, non-registered cursor over the Beacon's history.
// A Watcher belongs to exactly one goroutine; it is never touched by the
// producer or by other watchers.
type Watcher[V any] struct {
	b      *Beacon[V]
	cursor uint32 // absolute sequence of the NEXT value this watcher reads
	// no gen field: epoch arithmetic is derived on demand
	_ [4]byte
}

// Watch returns a Watcher positioned at the OLDEST retained value: full
// replay of all history. With cursor total values published and a ring of
// size S, the oldest retained value has sequence max(1, total-S+1).
func (b *Beacon[V]) Watch() *Watcher[V] {
	_, cur := unpack(b.claim.Load())
	hist := uint32(len(b.slots))
	w := &Watcher[V]{b: b}
	if cur > hist {
		w.cursor = cur - hist + 1 // cur values exist; keep the last hist
	}
	return w
}

// WatchLatest returns a Watcher positioned after the newest value: it sees
// only values published from this call onward.
func (b *Beacon[V]) WatchLatest() *Watcher[V] {
	_, cur := unpack(b.claim.Load())
	return &Watcher[V]{b: b, cursor: cur + 1}
}

// Cursor returns the watcher's position (absolute sequence of next read).
func (w *Watcher[V]) Cursor() uint32 { return w.cursor }

// Mode selects delivery semantics of Next.
type Mode int

const (
	// ModeBlock: deliver every value in order; block until the next value,
	// ctx.Done, or Beacon close. A watcher that laps gets a Skip event.
	ModeBlock Mode = iota
	// ModeLatest: coalesce lag — if the watcher laps, skip straight to the
	// newest value, reporting the skipped count with it.
	ModeLatest
	// ModeNonBlock: never wait. If a value is available return it; else
	// return ok=false with the lag amount (0 if caught up).
	ModeNonBlock
)

// Next advances the watcher by one value under the mode's semantics.
//
// In ModeBlock: blocks; returns (v, 0, true) per value in order; if the
// watcher lapped the ring, returns (zero, Skip, true) ONCE per lap event
// (the caller should call Next again to read the newest value); returns
// (zero, 0, false) on close (after which the watcher may still drain
// retained history via Replay) or ctx cancellation.
//
// In ModeLatest: never reports a lap as a separate event; lag coalesces
// into (newest v, skipped count, true).
//
// In ModeNonBlock: if a value is available at the cursor, return it; if
// the watcher lapped, return (zero, lag, false); if merely waiting, return
// (zero, 0, false).
//
// skip semantics: for ModeBlock, a nonzero skip IS the event (ok=true);
// for ModeLatest, skip accompanies the coalesced newest value; for
// ModeNonBlock with lag, skip is the lag and ok=false.
func (w *Watcher[V]) Next(mode Mode, ctx context.Context) (v V, skip Skip, ok bool) {
	b := w.b
	// Normalize the "0 == read from the very first value" sentinel into an
	// absolute sequence once, so all arithmetic below is uniform. (0 is the
	// only sequence that never exists, so it is safe as a sentinel.)
	if w.cursor == 0 {
		w.cursor = 1
	}
	for {
		// Re-read claim each iteration: gives us a monotonic-ish view.
		_, cur := unpack(b.claim.Load())

		// Values are sequences 1..cur (cur = newest sequence AND count of
		// values ever published; cursor is always >= 1 after the sentinel
		// normalization above). Unread values = [cursor, cur]; caught up
		// exactly when cursor == cur+1, i.e. delta == 0xFFFFFFFF.
		delta := cur - w.cursor
		if delta != 0xFFFFFFFF && delta < math.MaxUint32/2 {
			// Behind: unread count = delta+1 (>= 1).
			hist := uint32(len(b.slots))
			if delta+1 > hist {
				// Lapped: sequences [w.cursor, cur-hist] were overwritten.
				oldest := cur - hist + 1 // oldest retained sequence
				lap := oldest - w.cursor // count of lost values
				w.cursor = oldest
				// ALL modes coalesce the lap into the next returned value:
				// (newest v, Skip(lap), true). This keeps ModeBlock delivery
				// strict (the watcher yields the newest retained value
				// immediately) while still reporting exactly what was lost.
				skip = Skip(lap)
			}
			// Read the value at w.cursor (retry while torn).
			if v2, s2, ok2 := b.readSlotAt(w.cursor); ok2 {
				w.cursor++
				if mode == ModeLatest || mode == ModeNonBlock {
					return v2, skip, true
				}
				return v2, s2, true
			}
			// torn or overwritten mid-read: retry (bounded: writers are
			// wait-free, so the slot stabilizes quickly).
			runtime.Gosched()
			continue
		}

		// Cursor is at (or beyond) the published frontier: no value now.
		switch mode {
		case ModeNonBlock:
			return v, 0, false
		}
		// Blocking modes: gate wait.
		if !b.waitGate(&w.cursor, ctx) {
			if b.closed.Load() {
				return v, 0, false
			}
			return v, 0, false // ctx done
		}
	}
}

// readSlotAt reads the value with absolute sequence seq, retrying torn
// seqlock reads a bounded number of times. Returns ok=false if the slot's
// current occupant is NOT seq (overwritten) — the caller's lapping logic
// handles that case via the claim-word check.
func (b *Beacon[V]) readSlotAt(seq uint32) (V, Skip, bool) {
	idx := (seq - 1) & b.mask
	sp := &b.slots[idx]
	for attempts := 0; attempts < 1<<16; attempts++ {
		s := loadSlotSeq(sp) // atomic load; plain+fence on fast builds
		if s&1 == 1 {
			runtime.Gosched() // writer mid-write; resolve soon (wait-free writer)
			continue
		}
		var v V
		loadSlotVal(sp, &v)
		fenceAcquire()
		s2 := loadSlotSeq(sp)
		if s2 != s {
			var zero V
			return zero, 0, false // torn; retry
		}
		// Stable read. Validate that the slot's CURRENT occupant is exactly
		// `seq`: the value with sequence n lives in slot (n-1)&mask and the
		// slot's write counter after k writes is 2k, so the occupant's
		// sequence is (k-1)*size+idx+1 with k=s/2. A mismatch means our
		// value was overwritten (a lap) — the caller re-checks claim and
		// applies lapping. NOTE: torn-retry above must come FIRST; a torn
		// read also produces a mismatched occupant, and treating it as a
		// lap would double-count skips.
		size := uint32(len(b.slots))
		if held, valid := slotHolds(idx, uint32(s), size); !valid || held != seq {
			var zero V
			return zero, 0, false // lapped (or torn with equal s — impossible)
		}
		return v, 0, true
	}
	// Unreachable in practice (writers are wait-free); treat as lap.
	var zero V
	return zero, 0, false
}

// waitGate blocks until a value beyond *cursor is published, the Beacon
// closes, or ctx is done. Returns true when values may now be available
// (caller re-polls), false when the caller should return (closed/ctx).
func (b *Beacon[V]) waitGate(cursor *uint32, ctx context.Context) bool {
	for {
		_, cur := unpack(b.claim.Load())
		// Caught up == every published sequence 1..cur has been read, i.e.
		// the watcher's next-read cursor is cur+1 (cursor >= 1 always,
		// after Next's sentinel normalization).
		if cur != *cursor-1 {
			return true // value(s) available; re-poll
		}
		if b.closed.Load() {
			return false
		}
		g := b.wake.Load()
		if g == nil {
			// Beacon closed concurrently with wake teardown.
			return false
		}
		// The live gate is closed by the NEXT publish, which necessarily
		// produces a value beyond cursor-1 (the only way claim can advance
		// past cur). Park on it.
		// Register as waiter (for gate recycling + the parked fast path),
		// then select.
		b.parked.Add(1)
		atomic.AddInt32(&g.waiters, 1)
		if ctx == nil || ctx.Done() == nil {
			<-g.ch
			atomic.AddInt32(&g.waiters, -1)
			b.parked.Add(-1)
			return true
		}
		select {
		case <-g.ch:
			atomic.AddInt32(&g.waiters, -1)
			b.parked.Add(-1)
			return true
		case <-b.closingGate:
			atomic.AddInt32(&g.waiters, -1)
			b.parked.Add(-1)
			return b.claimAdvanceOrClosed(cursor)
		case <-ctx.Done():
			atomic.AddInt32(&g.waiters, -1)
			b.parked.Add(-1)
			return false
		}
	}
}

// claimAdvanceOrClosed reports whether waitGate's caller should re-poll
// (values arrived) or terminate (Beacon closed with nothing new).
func (b *Beacon[V]) claimAdvanceOrClosed(cursor *uint32) bool {
	_, cur := unpack(b.claim.Load())
	if cur != *cursor-1 {
		return true
	}
	return !b.closed.Load()
}

// Close terminates the Beacon: all blocked watchers wake and observe
// ok=false on their next Next; Publish returns ErrClosed; retained history
// remains readable (Replay) until all references drop. Idempotent,
// goroutine-safe, and leaks nothing.
func (b *Beacon[V]) Close() {
	if !b.closed.CompareAndSwap(false, true) {
		return
	}
	// Wake everyone: close the live wake gate AND the terminal closingGate
	// (watchers select on the wake gate; closing it releases them all).
	close(b.closingGate)
	b.pubMu.Lock()
	g := b.wake.Load()
	b.wake.Store(nil) // subsequent waitGate calls take the closed fast path
	b.pubMu.Unlock()
	if g != nil {
		close(g.ch)
	}
	// Release pooled gate objects.
	b.pubMu.Lock()
	b.pool.head = nil
	b.closedGates = nil
	b.pubMu.Unlock()
}

// Snapshot returns the newest value and the number of values ever
// published. ok=false if nothing was ever published.
func (b *Beacon[V]) Snapshot() (v V, total uint32, ok bool) {
	for {
		_, cur := unpack(b.claim.Load())
		if cur == 0 {
			return v, 0, false
		}
		// cur is the newest sequence (first value = 1); no -1 adjustment.
		if v2, _, ok2 := b.readSlotAt(cur); ok2 {
			return v2, cur, true
		}
		runtime.Gosched()
	}
}

// Len returns the number of retained values (0..capacity).
func (b *Beacon[V]) Len() int {
	_, cur := unpack(b.claim.Load())
	hist := uint32(len(b.slots))
	if cur > hist {
		return int(hist)
	}
	return int(cur)
}

// Capacity returns the history capacity (rounded-up power of two).
func (b *Beacon[V]) Capacity() int { return int(b.mask) + 1 }

// Closed reports whether Close was called.
func (b *Beacon[V]) Closed() bool { return b.closed.Load() }

// loadSlotSeq loads a slot's seq word with acquire semantics.
func loadSlotSeq[V any](sp *slot[V]) uint64 { return loadSlotSeqImpl(sp) }

// loadSlotVal copies the slot payload into dst with acquire semantics
// (tear-free is the caller's job via the seqlock re-check).
func loadSlotVal[V any](sp *slot[V], dst *V) { loadSlotValImpl(sp, dst) }

// storeSlotValImpl writes the payload with release semantics.
// (declared in beacon.go, dispatched here)
//   slotread_race.go  : all-atomic (TSan-friendly)
//   slotread_fast.go  : plain+fence  (needs !race)
//   slotread_plain.go : fallback when no .s fence exists (portable)

// slotSeqRaw provides the raw address of a slot's seq word for the atomic
// implementations in the build-tag files.
func slotSeqRaw[V any](sp *slot[V]) *uint64 {
	return (*uint64)(unsafe.Pointer(&sp.seq))
}

// unsafe.go helpers for payload word-copies:
func unsafeWordsOf[V any](v *V) (words []uintptr) {
	n := (int(unsafe.Sizeof(*v)) + wordSize - 1) / wordSize
	return unsafe.Slice((*uintptr)(unsafe.Pointer(v)), n)
}

const wordSize = int(unsafe.Sizeof(uintptr(0)))
